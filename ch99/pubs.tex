\chapter{Publications}
\label{sec:pubs}

This section lists all peer-reviewed publications resulting from the doctoral research undertaken during this thesis. The works are divided into two categories: main contributions directly within the scope of the thesis, followed by secondary collaborative works outside its core scope (ordered by degree of contribution).

\section{Directly related to this thesis}
\label{sec:pubs_first_author}

\begin{mybox}{WASPAA 2023}
    Enric Gusó, Joanna Luberadzka, Martí Baig, Umut Say\i n, \& Xavier Serra.
    \emph{An Objective Evaluation of Hearing Aids and DNN-Based Binaural Speech Enhancement in Complex Acoustic Scenes}.
    In: 2023 IEEE workshop on applications of signal processing to audio and acoustics (WASPAA), New Paltz, USA. 
\end{mybox}
\textbf{Contributions:} Training and evaluation of the models, generation of the binaural decoders, adaptation of the \acrshort{MASP} library, benchmark of the \acrshortpl{HA} recordings, writing of the manuscript, publication of the code and \acrshort{SOFA} files, and dissemination.

\begin{mybox}{WASPAA 2025}
    Enric Gusó, Joanna Luberadzka, Umut Say\i n, \& Xavier Serra.
    \emph{MB-RIRs: a Synthetic Room Impulse Response Dataset with Frequency-Dependent Absorption Coefficients}.
    In: 2025 IEEE workshop on applications of signal processing to audio and acoustics (WASPAA), Lake Tahoe, USA. 
\end{mybox}
\textbf{Contributions:} Adaptation of \acrshort{MASP} to multi-CPU processing. Implementation of source and receiver directivity in the \acrshort{ISM} algorithm. Generation of the dataset, deployment in an AWS environment for generation, training and evaluation of the models, writing of the manuscript, publication of the dataset and dissemination.

\begin{mybox}{ISMIR 2026}
    Enric Gusó \& Xavier Serra.
    \emph{CrowdioSet and PaRIRset: Two Datasets Towards Live Music Source Separation}.
    In: Proceedings of the 27th International Society for Music Information Retrieval Conference (ISMIR), Abu Dhabi, UAE. 
\end{mybox}
\textbf{Contributions:} Selection of the topic, data scraping and manually labeling the CrowdioSet dataset, recording and processing of the PaRIRset, training and evaluation of the \acrshort{RIR} ablations and the SCNet models, writing and revision of the manuscript, publication of the datasets, project webpage, dissemination.
